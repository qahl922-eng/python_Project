import hashlib
import os


def get_file_hash(filepath):
    """Generate SHA256 hash of a file."""
    
    hasher = hashlib.sha256()
    
    try:
        with open(filepath, "rb") as f:
            while True:
                chunk = f.read(4096)
                if not chunk:
                    break
                hasher.update(chunk)
    except Exception as e:
        print(f"Error reading file {filepath}: {e}")
        return None
    
    return hasher.hexdigest()


def compare_pdfs(path1, path2):
    """Compare two PDF files and return whether they are identical."""
    
    # Check existence
    if not os.path.isfile(path1):
        print(f"File not found or invalid: {path1}")
        return
    
    if not os.path.isfile(path2):
        print(f"File not found or invalid: {path2}")
        return
    
    # Check extension (case-insensitive)
    if not path1.lower().endswith(".pdf") or not path2.lower().endswith(".pdf"):
        print("Both files must be PDFs.")
        return
    
    # Generate hashes
    hash1 = get_file_hash(path1)
    hash2 = get_file_hash(path2)
    
    # Handle read errors
    if hash1 is None or hash2 is None:
        print("Could not compare files due to read error.")
        return
    
    # Compare
    if hash1 == hash2:
        print("✅ The two PDFs are identical.")
    else:
        print("❌ The two PDFs are different.")


# --- Run it ---
path1 = "file1.pdf"
path2 = "file2.pdf"

compare_pdfs(path1, path2)
